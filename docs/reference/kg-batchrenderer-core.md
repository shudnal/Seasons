```csharp
	public readonly struct BatchKey(int batchId, int cellX, int cellZ) : IEquatable<BatchKey>
	{
		public readonly int BatchId = batchId;

		public readonly int CellX = cellX;

		public readonly int CellZ = cellZ;

		public bool Equals(BatchKey other)
		{
			if (BatchId == other.BatchId && CellX == other.CellX)
			{
				return CellZ == other.CellZ;
			}
			return false;
		}

		public override bool Equals(object obj)
		{
			if (obj is BatchKey other)
			{
				return Equals(other);
			}
			return false;
		}

		public override int GetHashCode()
		{
			int batchId = BatchId;
			batchId = batchId * 397 + CellX;
			return batchId * 397 + CellZ;
		}
```

## Registration and swap removal

```csharp
	private static void DoRegister(BatchedInstance instance)
	{
		Matrix4x4 localToWorldMatrix = instance.transform.localToWorldMatrix;
		Vector3 position = localToWorldMatrix.GetPosition();
		int cellX = Mathf.FloorToInt(position.x * (1f / 32f));
		int cellZ = Mathf.FloorToInt(position.z * (1f / 32f));
		BatchKey key = new BatchKey(instance.BatchId, cellX, cellZ);
		if (!Buckets.TryGetValue(key, out var value))
		{
			value = new Bucket();
			Buckets[key] = value;
		}
		BatchDefinition batchDefinition = Definitions[instance.BatchId];
		int count = value.Matrices.Count;
		value.Matrices.Add(localToWorldMatrix * batchDefinition.LocalOffset);
		value.Instances.Add(instance);
		instance.SetBatch(key, count);
		instance.State = BatchedInstance.RegState.Registered;
	}

	private static void DoUnregister(BatchedInstance instance)
	{
		BatchKey batchKeyValue = instance.BatchKeyValue;
		int indexValue = instance.IndexValue;
		if (!Buckets.TryGetValue(batchKeyValue, out var value))
		{
			instance.State = BatchedInstance.RegState.Idle;
			return;
		}
		int num = value.Matrices.Count - 1;
		if (indexValue != num)
		{
			value.Matrices[indexValue] = value.Matrices[num];
			BatchedInstance batchedInstance = value.Instances[num];
			value.Instances[indexValue] = batchedInstance;
			batchedInstance.SetBatch(batchKeyValue, indexValue);
		}
		value.Matrices.RemoveAt(num);
		value.Instances.RemoveAt(num);
		if (value.Matrices.Count == 0)
		{
			Buckets.Remove(batchKeyValue);
		}
		instance.State = BatchedInstance.RegState.Idle;
	}
```

## Draw loop

```csharp
	public static void Draw()
	{
		foreach (KeyValuePair<BatchKey, Bucket> bucket in Buckets)
		{
			BatchKey key = bucket.Key;
			Bucket value = bucket.Value;
			int count = value.Matrices.Count;
			if (count != 0)
			{
				BatchDefinition batchDefinition = Definitions[key.BatchId];
				int num;
				for (int i = 0; i < count; i += num)
				{
					num = Mathf.Min(1023, count - i);
					Graphics.RenderMeshInstanced(in batchDefinition.RenderParams, batchDefinition.Mesh, 0, value.Matrices, num, i);
				}
			}
		}
	}
```
